falhas_por_area_com_variacao.sql 
-- Volume de falhas por área e semana, com variação vs. semana anterior.
-- Usa LAG para comparar sem self-join.

WITH base AS (
  SELECT
    area,
    DATE_TRUNC(data_falha, WEEK(MONDAY)) AS semana,
    COUNT(*) AS total
  FROM operacao.falhas
  WHERE data_falha >= DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY)
    AND status_registro = 'valido'
  GROUP BY area, semana
),

comparativo AS (
  SELECT
    area,
    semana,
    total,
    LAG(total) OVER (PARTITION BY area ORDER BY semana) AS total_anterior
  FROM base
)

SELECT
  area,
  semana,
  total,
  total_anterior,
  total - total_anterior AS delta,
  CASE
    WHEN total_anterior IS NULL  THEN 'sem base'
    WHEN total_anterior = 0       THEN 'novo'
    ELSE FORMAT('%.1f%%', SAFE_DIVIDE(total - total_anterior, total_anterior) * 100)
  END AS variacao
FROM comparativo
ORDER BY area, semana DESC;
filtrar_nao_processados.py 
# Nó de código no n8n — modo "Run Once for All Items".
# Entrada 0: registros da fonte. Entrada 1: IDs já processados.
# Devolve UM item não processado por execução (idempotência).

registros = _input.all(input_index=0)
processados = _input.all(input_index=1)

# conjunto de IDs já tratados — lookup O(1)
ids_feitos = set()
for p in processados:
    valor = p.json.get("id_registro")
    if valor:
        ids_feitos.add(str(valor).strip())

# primeiro registro válido e ainda não processado
for r in registros:
    dados = r.json
    rid = str(dados.get("id_registro", "")).strip()

    if not rid:
        continue
    if rid in ids_feitos:
        continue
    if not dados.get("categoria"):
        continue

    return dados

# nada pendente — sinaliza para o nó IF encerrar a rodada
return {"id_registro": "", "vazio": True}

// Estrutura do fluxo de relatório recorrente.
// Regra: uma execução processa um registro do início ao fim.

[Schedule Trigger]
      |
      +--> [Ler registros da fonte] ------+
      |                                    |
      +--> [Ler IDs processados] ---------+
                                           |
                                           v
                          [Code: filtrar_nao_processados]
                                           |
                                           v
                                    [IF vazio?]
                                     /        \
                                 true          false
                                   |             |
                              [No Operation]     v
                                            [Parser de campos]
                                                  |
                                                  v
                                          [Calcular métricas]
                                                  |
                                                  v
                                        [Gravar na base]
                                                  |
                                                  v
[Registrar ID processado]
                                                  |
                                                  v
                                       [Slack: enviar DM]
                                                  |
                                                  v
                                       [Atualizar métricas]

// Pontos de atenção que aprendi na prática:
//
// 1. "Registrar ID" vem DEPOIS de gravar, nunca antes.
//    Se inverter e a gravação falhar, o registro fica
//    marcado como processado sem ter entrado na base.
//
// 2. O IF precisa tratar o caso vazio explicitamente.
//    Sem isso o fluxo tenta processar objeto sem campo
//    e falha com erro que não diz nada.
//
// 3. Schedule curto + um item por execução é mais estável
//    que schedule longo + lote grande. Falha isolada não
//    derruba a rodada inteira.
//
// 4. Log estruturado em cada etapa crítica. Quando quebra
//    às 3h da manhã, o log é a única coisa que existe.
formulas-uteis.txt 
// Fórmulas que resolvem 80% do trabalho em base operacional.

// 1. Contagem por categoria com período móvel
=COUNTIFS(
    Registros!$C:$C; $A2;
    Registros!$A:$A; ">=" & EOMONTH(TODAY(); -1) + 1;
    Registros!$E:$E; "valido"
)

// 2. Consolidar múltiplas abas sem copiar e colar
=QUERY(
    {Area_A!A2:F; Area_B!A2:F; Area_C!A2:F; Area_D!A2:F};
    "select Col1, Col3, count(Col1)
     where Col1 is not null
     group by Col1, Col3
     order by count(Col1) desc
     label count(Col1) 'total'";
    0
)

// 3. Semana ISO — evita divergência de contagem entre relatórios
=ISOWEEKNUM($A2) & "/" & YEAR($A2)

// 4. Detectar duplicidade antes que ela vire métrica errada
=IF(COUNTIF($B$2:$B2; $B2) > 1; "DUPLICADO"; "")

// 5. Variação percentual protegida contra divisão por zero
=IFERROR(
    TEXT((C2 - B2) / B2; "0.0%");
    IF(C2 > 0; "novo"; "—")
)

// 6. Ranking de concentração (regra de Pareto na operação)
=ARRAYFORMULA(
    IF(B2:B <> "";
       ROUND(SUMIF(ROW(B2:B); "<=" & ROW(B2:B); B2:B) / SUM(B2:B) * 100; 1);
       "")
)
Princípios

Regras que sigo
ao escrever automação.

01

Idempotência primeiro

Todo fluxo que grava precisa poder rodar duas vezes sem duplicar. Sem isso, a primeira reexecução destrói a confiança na métrica.

02

Falhar alto e claro

Erro silencioso é pior que fluxo parado. Registro inválido fica marcado e visível, nunca descartado sem rastro.

03

Um item por execução

Lote grande é tentador e frágil. Processar unitário custa tempo e devolve previsibilidade.

04

Categoria fechada

Texto livre em campo de classificação inviabiliza agrupamento. Se o dado precisa virar número, a entrada precisa ser controlada.

05

Entregar onde a pessoa está

Relatório no Slack vence dashboard em link. A melhor visualização é a que aparece sem ser procurada.

06

Log é infraestrutura

Quando o fluxo quebra fora do horário, o log é a única coisa disponível. Escrever log é parte da feature, não extra.
